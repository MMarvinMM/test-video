    row record;

		EXECUTE format('CREATE SEQUENCE IF NOT EXISTS %I_gid_seq
            INCREMENT 1
            START 1
            MINVALUE 1
            MAXVALUE 2147483647
            CACHE 1;',replace(row.tablename, 'polygon_xztb','polygon_xztb_bak'));
		EXECUTE format('insert into %I select * from %I;',replace(row.tablename, 'polygon_xztb','polygon_xztb_bak'), row.tablename);
    END LOOP;
END $$;
